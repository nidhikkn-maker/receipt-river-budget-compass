// P0-011 repro (before fix): deleting a company profile drops the profile, its tax years and account links.
import { afterEach, beforeEach, describe, expect, it, vi } from "vitest";
import { cleanup, fireEvent, render, screen } from "@testing-library/react";
import { harness, reset } from "./harness";

vi.mock("@/lib/store", async () => ({ useStore: (await import("./harness")).useStoreMock }));
vi.mock("@tanstack/react-router", async () => (await import("./harness")).routerMock);
vi.mock("@/components/AppShell", () => ({ PageTitle: ({ title }: { title: string }) => <h1>{title}</h1> }));
vi.mock("@/components/books/CompanyTaxProfile", () => ({ CompanyTaxProfile: () => null }));
vi.mock("@/components/books/CompanyAccess", () => ({ CompanyAccess: () => null, SharedWithMe: () => null }));
vi.mock("@/lib/confirm", () => ({ confirmAction: vi.fn(async () => true) }));
import { Route } from "@/routes/_authenticated/entities";
import { confirmAction } from "@/lib/confirm";
const Entities = (Route as unknown as { component: () => React.ReactElement }).component;

const co = { id: "ent-1", name: "Thapal Ventures Pvt Ltd", kind: "company", pan: "AAACT1234F", gstins: ["33AAACT1234F1Z5"], booksFrom: "2024-25" };
const years = [{ id: "y1", entityId: "ent-1", fy: "2025-26", bookProfit: 420000 }];
beforeEach(() => reset({ entities: [co], books: { entityAccounts: { "ent-1": ["bank-1"] }, entityTaxProfiles: years } }, { accounts: [{ id: "bank-1", name: "HDFC current", kind: "current", openingBalance: 0, archived: false }] }));
afterEach(cleanup);

describe("P0-011 repro: company profiles", () => {
  it("Delete (generic yes/no, no reason) removes the profile, its tax years and its account links", async () => {
    render(<Entities />);
    fireEvent.click(screen.getByRole("button", { name: "Delete" }));
    await vi.waitFor(() => expect(harness.saves).toBe(1));
    expect(confirmAction).toHaveBeenCalledWith(expect.objectContaining({ title: "Delete this profile?" }));
    const books = harness.stored["books"] as Record<string, unknown>;
    expect(harness.stored["entities"]).toEqual([]);
    expect(books["entityTaxProfiles"]).toEqual([]);
    expect(books["entityAccounts"]).toEqual({});
  });
});
