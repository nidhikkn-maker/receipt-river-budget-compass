// P0-012 repro (before fix): vault documents are hard-deleted (row and file) behind a generic confirm.
import { readFileSync } from "node:fs";
import { afterEach, beforeEach, describe, expect, it, vi } from "vitest";
import { cleanup, fireEvent, render, screen } from "@testing-library/react";
import { callsOf, reset, spy } from "./harness";

vi.mock("@/lib/store", async () => ({ useStore: (await import("./harness")).useStoreMock }));
vi.mock("@tanstack/react-router", async () => (await import("./harness")).routerMock);
vi.mock("@/components/AppShell", () => ({ PageTitle: ({ title }: { title: string }) => <h1>{title}</h1> }));
vi.mock("@/components/ContractObligations", () => ({ ContractObligations: () => null, NoticeDeadlines: () => null }));
vi.mock("@/components/books/VaultBooks", () => ({ VaultBooks: () => null }));
vi.mock("@/components/ExpenseDialog", () => ({ ExpenseDialog: () => null }));
vi.mock("@/lib/confirm", () => ({ confirmAction: vi.fn(async () => true) }));
import { Route } from "@/routes/_authenticated/vault";
const Vault = (Route as unknown as { component: () => React.ReactElement }).component;

const doc = { id: "doc-1", title: "LIC policy 2026", docType: "insurance", filePath: "owner/docs/lic.pdf", fileName: "lic.pdf", mimeType: "application/pdf", size: 1000, docDate: "2026-04-01", amount: 24000, currency: "INR", merchant: "LIC", notes: "", text: "", transactionId: null, accountId: null, loanId: null, expiresOn: null, createdAt: "2026-04-02T00:00:00Z", fields: [] };
beforeEach(() => reset({}, { documents: [doc], removeDocument: spy("removeDocument"), updateDocument: spy("updateDocument"), signedUrl: async () => null, downloadFile: async () => undefined }));
afterEach(cleanup);

describe("P0-012 repro: vault documents", () => {
  it("Delete calls store.removeDocument after a generic confirm, with no reason", async () => {
    render(<Vault />);
    fireEvent.click(screen.getByRole("button", { name: `Delete document ${doc.title}` }));
    await vi.waitFor(() => expect(callsOf("removeDocument")).toHaveLength(1));
  });
  it("store.removeDocument deletes the documents row and the stored file", () => {
    const src = readFileSync("src/lib/store.tsx", "utf8");
    const body = src.slice(src.indexOf("const removeDocument"), src.indexOf("const removeDocument") + 900);
    expect(body).toMatch(/from\("documents"\)\.delete\(\)/);
    expect(body).toMatch(/storage\.from\("receipts"\)\.remove/);
  });
});
