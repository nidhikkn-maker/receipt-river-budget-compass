// P0-004 / P0-007 reproduction: the real StoreProvider (src/lib/store.tsx) with a fake Supabase client whose
// ledger (expenses) and preferences reads fail, after the encrypted device copy was painted.
// Proves the CURRENT behaviour: a failed read replaces the last known records with empty lists, reports
// "synced", and writes the empty lists back over the device copy.
import { act, cleanup, render } from "@testing-library/react";
import { afterEach, describe, expect, it, vi } from "vitest";

const h = vi.hoisted(() => ({
  fail: new Set<string>(),
  snapshot: null as unknown,
  put: [] as { expenses: unknown[]; preferences: { trips: unknown[] } }[],
}));
vi.mock("@/integrations/supabase/client", () => {
  const result = (table: string) => h.fail.has(table)
    ? { data: null, error: { message: "Could not connect to the database", code: "PGRST000" } }
    : { data: table === "finance_preferences" || table === "profiles" ? null : [], error: null };
  const builder = (table: string): unknown => new Proxy({}, {
    get: (_t, k) => k === "then" ? (ok: (v: unknown) => void, bad: (e: unknown) => void) => Promise.resolve(result(table)).then(ok, bad)
      : k === "maybeSingle" || k === "single" ? () => Promise.resolve(result(table)) : () => builder(table),
  });
  return { supabase: { from: (t: string) => builder(t), rpc: async () => ({ data: null, error: null }), storage: { from: () => ({ remove: async () => ({ error: null }) }) } } };
});
vi.mock("./offline", () => ({
  getSnapshot: async () => h.snapshot,
  putSnapshot: async (_o: string, s: { expenses: unknown[]; preferences: { trips: unknown[] } }) => { h.put.push(s); },
  pendingEntries: async () => [], queueEntry: async () => {}, removePending: async () => {},
}));
vi.mock("./fx.functions", () => ({ saveExpense: async () => ({ ok: true }) }));
vi.mock("sonner", () => ({ toast: Object.assign(vi.fn(), { error: vi.fn(), success: vi.fn(), message: vi.fn(), info: vi.fn() }) }));

import { StoreProvider, useStore } from "./store";
import { defaultPreferences } from "./preferences";

type Seen = ReturnType<typeof useStore>;
const seen: { s: Seen | null } = { s: null };
function Probe() { seen.s = useStore(); return null; }
const cachedExpense = { id: "e1", title: "Swiggy order", amount: 640, originalAmount: 640, currency: "INR", fxRate: 1, date: "2026-11-01", category: "Food", scope: "personal", txnType: "expense", accountId: "bank", receipts: [], tags: [], splits: [], allocations: [], paymentMethod: "upi" };
const trip = { id: "t1", name: "Goa", start: "2026-12-01", end: "2026-12-05" };

afterEach(() => { cleanup(); h.fail.clear(); h.put.length = 0; seen.s = null; });

describe("P0-007 REPRO: a failed ledger/preferences read wipes the shown records and the device copy", () => {
  it("replaces cached transactions and trips with empty lists, reports synced, and persists the empty copy", async () => {
    Object.defineProperty(navigator, "onLine", { value: true, configurable: true });
    h.snapshot = { name: "KK", members: [], expenses: [cachedExpense], accounts: [{ id: "bank", name: "HDFC", kind: "savings", openingBalance: 1000, archived: false }], commitments: [], rawLoans: [], goals: [], budgets: {}, preferences: { ...defaultPreferences(), trips: [trip] }, valuations: [], documents: [] };
    h.fail.add("expenses"); h.fail.add("finance_preferences");
    await act(async () => { render(<StoreProvider userId="11111111-1111-4111-8111-111111111111"><Probe /></StoreProvider>); });
    await act(async () => { await new Promise((r) => setTimeout(r, 50)); });
    const s = seen.s!;
    expect(s.transactions).toHaveLength(0);          // the cached Swiggy entry is gone from every screen
    expect(s.preferences.trips).toHaveLength(0);     // trips (and every other preference) reset to defaults
    expect(s.syncState).toBe("synced");              // header says up to date
    expect(h.put.at(-1)?.expenses).toEqual([]);      // the empty list overwrote the device copy
    expect(h.put.at(-1)?.preferences.trips).toEqual([]);
  });
});
